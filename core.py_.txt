#!/usr/bin/env python3
"""The Saem door from Python, plus the runtime ledger.

The core holds no state, so the state lives here: a blackboard Json, an
entropy-debt counter, and an append-only ledger of every overwrite --
the machine cousin of THINK.md §3. Replay the ledger and the board and
the debt come back exactly; a board file on disk is only a cache.

    from saem.core import Core
    with Core() as c:
        r = c.result("gen.make", id="K1", seed=7)
        board, debt = c.set({}, 0.0, "receipt", r, "ledger.jsonl")
"""
import json
import os
import subprocess
import sys
import time

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
CLI = os.path.join(ROOT, "build", "saem_cli")
BUILD = os.path.join(ROOT, "tools", "build.py")


class SourceIdMismatch(SystemExit):
    """The binary's body is not the tree's body. Refuse to work."""


class Core:
    """One child process, the line protocol, source-id checked at spawn."""

    def __init__(self, cli=CLI, require_source=True):
        if require_source:
            disk = subprocess.run([sys.executable, BUILD, "id"],
                                  capture_output=True, text=True).stdout.strip()
            body = subprocess.run([cli, "--source-id"],
                                  capture_output=True, text=True).stdout.strip()
            if disk != body:
                raise SourceIdMismatch(
                    "binary body %s != tree %s; run: python3 tools/build.py cli"
                    % (body, disk))
        self.p = subprocess.Popen([cli], stdin=subprocess.PIPE,
                                  stdout=subprocess.PIPE, text=True, bufsize=1)
        self._id = 0

    def call(self, method, **args):
        self._id += 1
        self.p.stdin.write(json.dumps(
            {"id": self._id, "method": method, "args": args},
            ensure_ascii=False) + "\n")
        self.p.stdin.flush()
        line = self.p.stdout.readline()
        if not line:
            raise SystemExit("core closed the door")
        env = json.loads(line)
        if env.get("id") != self._id:
            raise SystemExit("core answered a different question")
        return env

    def result(self, method, **args):
        env = self.call(method, **args)
        if not env.get("ok"):
            raise ValueError(env.get("error", "unknown failure"))
        return env["result"]

    def close(self):
        try:
            self.p.stdin.close()
            self.p.wait(timeout=10)
        except Exception:
            self.p.kill()

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()
        return False

    # ---- the ledger: blackboard + Landauer debt, append-only ----------
    def set(self, board, debt, key, value, ledger=None):
        """Pure kernel.set through the door; the overwrite is recorded."""
        r = self.result("kernel.set", board=board, debt=debt,
                        key=key, value=value)
        if ledger:
            append(ledger, {"ts": time.time(), "op": "set", "key": key,
                            "value": value, "debt": r["debt"], "soul": r["soul"]})
        return r["board"], r["debt"]

    def replay(self, ledger):
        """Crash recovery: board and debt from the ledger alone."""
        board, debt = {}, 0.0
        for e in read(ledger):
            if e.get("op") == "set":
                board[e["key"]] = e["value"]
                debt = e["debt"]
        return board, debt


def append(path, entry):
    with open(path, "a", encoding="utf-8") as f:
        f.write(json.dumps(entry, ensure_ascii=False) + "\n")


def read(path):
    if not os.path.exists(path):
        return []
    out = []
    with open(path, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                out.append(json.loads(line))
    return out
